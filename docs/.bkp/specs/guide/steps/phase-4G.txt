TYPES (new — to be propagated to orchestrator.txt after approval)

  enum ExecutionMode {
      Interactive,
      Autonomous,
  }
  // Already defined in orchestrator::types (phase-4F:L161). Reused here.

  struct AutonomousGate {
      inner: Box<dyn InteractiveGate>,       // wraps existing HITL gate
      evaluator: AutoAdvanceEvaluator,       // checks evidence artifacts
      mode: ExecutionMode,                   // per-session, from SessionState
      escalation_policy: EscalationPolicy,   // configurable thresholds
      dispatch_table: PlanDispatchTable,     // dynamic dispatch from PipelineSpec (technical-analysis-v3 §5)
  }
  // AutonomousGate implements the same InteractiveGate trait.
  // In Interactive mode: delegates to inner directly (zero behavior change).
  // In Autonomous mode: evaluator checks evidence first, delegates to inner (human pause) only
  //   on HIGH-risk gate or missing/invalid evidence. Within EXECUTING state, dangerous tool
  //   calls (file deletion, etc.) are reviewed by auditor agent, not human (arch-overview S6:L220).
  //   This is a different path from gate transitions — tool-call review is in-state, not at gate.
  // AutonomousGate uses DispatchPipeline/PlanDispatchTable types for agent dispatch.
  //   Plan-dispatch states (TESTING_FIRST, EXECUTING, AUDITING, MEMORIZING) use
  //   PlanDispatchTable built from PipelineSpec at L2. Meta-dispatch states (PLANNING,
  //   CONFIGURING) use static MetaDispatchTable. Source: technical-analysis-v3.txt §5.

  struct AutoAdvanceEvaluator {
      gate_checks: HashMap<FsmTransition, EvidenceCheck>,
      // Maps each FSM transition to its required evidence artifact check.
      // Populated from evidence gate protocol (orchestrator.txt S13):
      //   IDLE->PLANNING: knowledge_engine.readiness_check() != UNAVAILABLE
      //   PLANNING->TESTING_FIRST: grounding_sweep_report has zero UNGROUNDED
      //   TESTING_FIRST->EXECUTING: all tests fail for right reasons. Heuristic:
      //     cargo test exit code != 0 (tests ran) AND zero "error[E" in stderr
      //     (no compilation errors — compilation errors are "wrong reasons").
      //     This is an approximation: a test that panics during setup looks like
      //     a "right reason" failure. Acceptable for auto-advance because AUDITING
      //     catches false positives. If too many false advances, tighten heuristic.
      //   EXECUTING->AUDITING: at least 1 test flipped FAIL->PASS (diff test results)
      //   AUDITING->MEMORIZING: zero BLOCK findings in audit_report
      //   MEMORIZING->next: DKG updated AND changelog has entry
  }

  struct EvidenceCheck {
      artifact_path: String,                 // relative to .elai_cc/sessions/{sid}/
      check_fn: fn(&[u8]) -> EvidenceVerdict,  // deterministic, no LLM
      description: String,                   // for audit trail
  }

  enum EvidenceVerdict {
      Pass,                                  // artifact present and valid -> auto-advance
      Fail { reason: String },               // artifact invalid -> block (same as INTERACTIVE)
      Missing,                               // artifact not found -> block
  }

  struct EscalationPolicy {
      anti_spiral_threshold: u32,            // default 3 — triggers on 3rd failure (orchestrator.txt
                                             // S7:L180 "3x same GRC step failure"). Note: L173 says
                                             // "same_step_count > 3" (4th). We pin to L180 (3x).
                                             // Spec inconsistency documented in grounding log.
      high_risk_pauses_in_both_modes: bool,  // default true (architecture-overview.txt S6:L227
                                             // "HIGH: pause in both modes"). Note: L220 says
                                             // "dangerous actions reviewed by auditor agent, not human"
                                             // — this refers to tool-call review within EXECUTING,
                                             // not FSM gate transitions. Gates pause for human.
      replan_major_always_human: bool,       // default true (orchestrator.txt S13:L590)
      max_auto_advances_per_phase: u32,      // circuit breaker: if >N auto-advances without
                                             // human check, force one pause. Default 10.
  }

  enum EscalationTrigger {
      AntiSpiral { step_id: String, count: u32 },        // same GRC step failed 3x
      HighRisk { transition: FsmTransition },             // HIGH-risk gate
      ReplanMajor { evidence: String },                   // ReplanRequest(MAJOR)
      CircuitBreaker { auto_advance_count: u32 },         // max_auto_advances_per_phase exceeded
      SubstituteModelFailed { tried: Vec<String> },       // no fallback model fits
      SplitTaskFailed { reason: String },                 // decomposition produced 0 sub-tasks
  }

  struct SplitTaskPlan {
      parent_phase: String,                  // phase that exhausted context budget
      sub_tasks: Vec<SubTask>,               // max 4 sub-tasks (prevent exponential blowup)
      merge_strategy: MergeStrategy,         // how sub-task results rejoin
      max_depth: u32,                        // 2 (no recursive splits — sub-task exhaustion -> ESCALATE)
      current_depth: u32,                    // tracked per session
  }

  struct SubTask {
      task_id: String,
      scope: String,                         // subset of parent PipelineSpec
      context_budget: usize,                 // fresh budget per sub-task
      depends_on: Vec<String>,               // ordering constraints (DAG, not tree)
  }

  enum MergeStrategy {
      Sequential,                            // sub-tasks execute in order, each writes to parent
                                             // session's SSB directly. No branching needed.
                                             // NOTE: S10 branching (orchestrator.txt L312) has
                                             // pick-one semantics ("pick winning branch, discard
                                             // others") — incompatible with SPLIT_TASK which needs
                                             // ALL sub-task results. Sequential avoids this.
                                             // Future: if parallel sub-tasks needed, define a new
                                             // merge protocol (not S10 branches). Deferred.
  }

  struct ModelFallback {
      role: String,                          // agent role that exceeded max_restarts
      current_model: String,                 // model that failed
      fallback_chain: Vec<String>,           // from elai_cc.toml [agents.{role}.fallback_models]
      attempted: Vec<String>,                // models already tried this phase
  }
  // Selection: iterate fallback_chain, skip attempted, use ModelPool swap protocol
  //   (providers.txt S4:L111-115): check VRAM via "Fits alongside current?" multiplexing
  //   check or unload-then-load path. ModelPool has can_fit_onnx() for ONNX models but
  //   no can_fit() for LLM models — VRAM feasibility uses the swap protocol's implicit
  //   check (step 2: "Fits alongside current?" / step 3: "wait for checkpoint, unload, load").
  //   NOTE: Phase 4G MUST add a can_fit_llm(model_id) -> bool method to ModelPool
  //   (new API — propagate to providers.txt after approval). Without it, SUBSTITUTE_MODEL
  //   cannot pre-check VRAM before attempting load (load failure = wasted swap time).
  // If no model fits VRAM -> EscalationTrigger::SubstituteModelFailed.
  // Fallback chain example: ["gemma4-31b-q4", "gemma4-12b-q4", "gemma4-4b-q8"]
  // SLO tier mismatch logged as WARNING (providers.txt S4:L85) but allowed.

        4. SPLIT_TASK: on context budget Terminal, decompose PipelineSpec phases into
           sub-tasks. Each sub-task: fresh SessionState, fresh context budget, own SSB branch.
           Max 4 sub-tasks. Max depth 2. Sub-tasks run sequentially (not parallel — simpler,
           and budget exhaustion implies resource pressure). Each sub-task writes directly to
           parent session's SSB (Sequential merge — no S10 branching, which has pick-one semantics).
        5. SUBSTITUTE_MODEL: on max_restarts, iterate fallback_chain.
           For each candidate: ModelPool::can_fit_llm(candidate) -> if true, swap protocol
           (providers.txt S4:L111-115: unload current, load new, health check).
           If none fit VRAM -> ESCALATE. (can_fit_llm is new API added by this phase.)
        6. Mode switch: SessionState.execution_mode field. CLI command: /mode autonomous
           or /mode interactive. Takes effect at next gate evaluation (not mid-phase).
           TOCTOU prevention: gate evaluation reads mode under Arc<RwLock<SessionState>>
           read lock; mode switch acquires write lock. Standard Rust RwLock semantics —
           no custom locking primitive needed. NOTE: SessionState is not yet a defined
           struct in any spec — Phase 3A or 3B must define it. Phase 4G adds the
           execution_mode field to whatever struct exists by then.
        7. Evidence provenance: EvidenceCheck includes expected_producer: String (role name,
           e.g. "auditor", "executor" — matches role field from AgentPool::spawn()).
           AutoAdvanceEvaluator verifies artifact was written by correct agent: query SSB
           for the artifact's seq, join to get agent_id, look up agent_id in AgentPool's
           internal registry (HashMap<Uuid, AgentHandle> from orchestrator.txt S12) to get
           role. NOTE: AgentPool currently has spawn/spawn_parallel/communicate — no role
           lookup. Phase 4G MUST add fn get_role(agent_id: Uuid) -> Option<String> to
           AgentPool (new API — propagate to orchestrator.txt after approval).

TREE-LEVEL AUTONOMOUS ADVANCE EXTENSION

  The base 4G auto-advances at FSM gate transitions (phase-level).
  This extension auto-advances at the baby step level within the DecompositionTree.

  DESIGN
    In AUTONOMOUS mode with a DecompositionTree, the system auto-advances through
    baby steps without human intervention, subject to the same evidence checks.
    The granularity is finer: instead of one human gate per FSM phase, there is
    one evidence check per baby step cycle (TESTING_FIRST->EXECUTING->AUDITING).

    Per-baby-step auto-advance:
      TESTING_FIRST->EXECUTING: tests compile and fail -> auto-advance
      EXECUTING->AUDITING: at least 1 test flipped -> auto-advance
      AUDITING->ACCEPTANCE: zero BLOCKs -> auto-advance to acceptance check
      ACCEPTANCE->NEXT_BABY_STEP: all acceptance criteria met -> TreeNavigator advances

    Escalation within tree:
      Baby step failure after max retries -> BubbleUpHandler (4E) determines action
      BubbleUp to step level -> re-research (auto in AUTONOMOUS, human in INTERACTIVE)
      BubbleUp to phase level -> INTERACTIVE: human decides. AUTONOMOUS: auto if LOW risk.
      BubbleUp to concept level -> always human (HIGH risk, fundamental approach change)

    Integration auto-advance:
      BABY_STEP integration pass -> auto-advance to next baby step
      BABY_STEP integration fail -> auto-retry once, then bubble up
      STEP integration pass -> auto-advance to next step
      PHASE integration pass -> auto-advance to next phase
      CONCEPT integration pass -> auto-advance to next concept
      CONCEPT integration fail -> always human (end-to-end failure)

  NEW TYPES

    struct TreeAutoAdvancePolicy {
        baby_step_auto: bool,                    // auto-advance per baby step (default true in AUTONOMOUS)
        step_integration_auto: bool,             // auto-advance on step integration (default true)
        phase_integration_auto: bool,            // auto-advance on phase integration (default true)
        concept_integration_auto: bool,          // auto-advance on concept integration (default false — always human)
        max_auto_baby_steps_per_concept: u32,    // circuit breaker (default 50)
        bubble_up_risk: BTreeMap<DecompositionLevel, RiskLevel>,
          // BabyStep -> LOW, Step -> LOW, Phase -> MEDIUM, Concept -> HIGH
    }

    Merges with existing EscalationPolicy. TreeAutoAdvancePolicy extends it for tree context.
